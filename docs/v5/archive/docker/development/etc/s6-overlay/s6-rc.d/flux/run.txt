#!/bin/sh

cd /var/www/html

. /etc/s6-overlay/scripts/container-role
role="$(xlaunch_container_role_value)"

if ! xlaunch_container_has_role flux; then
    echo "   INFO  Flux is disabled for role '$role', sleeping."
    exec sleep infinity
fi

if grep -qE '^XLAUNCH_FLUX_ENABLED=false' .env 2>/dev/null || [ "${XLAUNCH_FLUX_ENABLED:-}" = "false" ]; then
    echo "   INFO  Flux is disabled, sleeping."
    exec sleep infinity
fi

export XLAUNCH_FLUX_GRPC_BIND="${XLAUNCH_FLUX_GRPC_BIND:-0.0.0.0:6443}"
export XLAUNCH_FLUX_UNIX_SOCKET_PATH="${XLAUNCH_FLUX_UNIX_SOCKET_PATH:-/run/xlaunch/flux.sock}"
export XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH="${XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH:-/var/www/html/storage/app/flux/jwt.priv}"
export XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH="${XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH:-/var/www/html/storage/app/flux/jwt.pub}"
export XLAUNCH_FLUX_ALLOW_PUBLIC_BIND="${XLAUNCH_FLUX_ALLOW_PUBLIC_BIND:-1}"
export XLAUNCH_FLUX_REQUIRE_HOST_BINDING="${XLAUNCH_FLUX_REQUIRE_HOST_BINDING:-0}"
export XLAUNCH_FLUX_LARAVEL_API_URL="${XLAUNCH_FLUX_LARAVEL_API_URL:-http://127.0.0.1:8080}"
if [ -z "${XLAUNCH_FLUX_LARAVEL_API_TOKEN:-}" ] && [ -f .env ]; then
    XLAUNCH_FLUX_LARAVEL_API_TOKEN="$(grep -E '^XLAUNCH_FLUX_LARAVEL_API_TOKEN=' .env 2>/dev/null | tail -n1 | cut -d= -f2- | sed "s/^['\"]//; s/['\"]$//")"
fi
export XLAUNCH_FLUX_LARAVEL_API_TOKEN="${XLAUNCH_FLUX_LARAVEL_API_TOKEN:-}"
if [ ! -r "$XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH" ]; then
    echo "   INFO  Flux JWT public key not found at $XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH, generating keypair..."
    mkdir -p "$(dirname "$XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH")" "$(dirname "$XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH")"
    openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out "$XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH.tmp"
    chmod 0600 "$XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH.tmp"
    mv "$XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH.tmp" "$XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH"
    openssl pkey -in "$XLAUNCH_FLUX_JWT_PRIVATE_KEY_PATH" -pubout -out "$XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH.tmp"
    chmod 0644 "$XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH.tmp"
    mv "$XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH.tmp" "$XLAUNCH_FLUX_JWT_PUBLIC_KEY_PATH"
fi

if ! /usr/local/bin/flux --version >/dev/null 2>&1; then
    echo "   ERROR  Flux binary cannot run in this container. Check that the installed coold nightly Flux artifact is compatible with Alpine."
    exec sleep infinity
fi

mkdir -p "$(dirname "$XLAUNCH_FLUX_UNIX_SOCKET_PATH")"

echo "   INFO  Flux is enabled for role '$role', starting..."
exec /usr/local/bin/flux
